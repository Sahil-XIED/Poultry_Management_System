MySQL Database Schema
Run these queries in phpMyAdmin or MySQL CLI to set up the backend
🛢️ Full Schema — poultry_farm database
-- FARMGUARD — poultry_farm schema CREATE DATABASE IF NOT EXISTS poultry_farm CHARACTER SET utf8mb4; USE poultry_farm; CREATE TABLE IF NOT EXISTS flock ( id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(100) NOT NULL, breed VARCHAR(100) NOT NULL, purpose ENUM('Layer','Broiler','Breeder','Dual Purpose') NOT NULL, initial_birds INT DEFAULT 0, current_birds INT DEFAULT 0, age_days INT DEFAULT 0, placed_on DATE NOT NULL, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ) ENGINE=InnoDB; CREATE TABLE IF NOT EXISTS mortality ( id INT AUTO_INCREMENT PRIMARY KEY, flock_id INT NOT NULL, death_date DATE NOT NULL, death_count INT NOT NULL, cause VARCHAR(100) DEFAULT 'Unknown', notes TEXT, severity ENUM('Normal','Warning','Critical') DEFAULT 'Normal', FOREIGN KEY (flock_id) REFERENCES flock(id) ON DELETE CASCADE ) ENGINE=InnoDB; CREATE TABLE IF NOT EXISTS disease_log ( id INT AUTO_INCREMENT PRIMARY KEY, observed_on DATE NOT NULL, disease_name VARCHAR(100) NOT NULL, birds_affected INT DEFAULT 0, severity ENUM('Mild','Moderate','Severe','Critical'), symptoms TEXT, treatment TEXT, resolved TINYINT DEFAULT 0 ) ENGINE=InnoDB; CREATE TABLE IF NOT EXISTS vaccination ( id INT AUTO_INCREMENT PRIMARY KEY, flock_id INT NOT NULL, vaccine_name VARCHAR(100) NOT NULL, birds_vaccinated INT NOT NULL, admin_date DATE NOT NULL, next_due DATE, method VARCHAR(50), status ENUM('Done','Due','Overdue') DEFAULT 'Done', FOREIGN KEY (flock_id) REFERENCES flock(id) ON DELETE CASCADE ) ENGINE=InnoDB; CREATE TABLE IF NOT EXISTS feed_log ( id INT AUTO_INCREMENT PRIMARY KEY, flock_id INT NOT NULL, log_date DATE NOT NULL, feed_kg DECIMAL(8,2) NOT NULL, water_litres DECIMAL(8,1) NOT NULL, feed_type VARCHAR(50), stock_after DECIMAL(8,2), FOREIGN KEY (flock_id) REFERENCES flock(id) ON DELETE CASCADE ) ENGINE=InnoDB; CREATE TABLE IF NOT EXISTS egg_production ( id INT AUTO_INCREMENT PRIMARY KEY, flock_id INT NOT NULL, collection_date DATE NOT NULL, total_eggs INT DEFAULT 0, broken_eggs INT DEFAULT 0, lay_rate DECIMAL(5,2), FOREIGN KEY (flock_id) REFERENCES flock(id) ON DELETE CASCADE ) ENGINE=InnoDB;
